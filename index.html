document.getElementById('searchBtn').addEventListener('click', async function() { const uid = document.getElementById('uidInput').value.trim(); const loader = document.getElementById('loader'); const errorMsg = document.getElementById('errorMsg'); const resultCard = document.getElementById('resultCard'); if (!uid) { alert("Please enter a valid UID"); return; } resultCard.style.display = 'none'; errorMsg.style.display = 'none'; loader.style.display = 'block'; const targetUrl = 'https://ff-info-ro45.vercel.app/api?uid=' + uid + '&key=Anurag'; const proxyUrl = 'https://api.allorigins.win/get?url=' + encodeURIComponent(targetUrl); try { const response = await fetch(proxyUrl); if (!response.ok) throw new Error("Network Response Error"); const wrapper = await response.json(); const data = JSON.parse(wrapper.contents); if (data && data.basicInfo) { document.getElementById('pName').innerText = data.basicInfo.nickname || "N/A"; document.getElementById('pUid').innerText = data.basicInfo.accountId || uid; document.getElementById('pLevel').innerText = data.basicInfo.level || "0"; document.getElementById('pLikes').innerText = data.basicInfo.liked || "0"; document.getElementById('pRegion').innerText = data.basicInfo.region || "N/A"; document.getElementById('pRank').innerText = data.basicInfo.rankingPoints || "0"; document.getElementById('pCsRank').innerText = data.basicInfo.csRank || "0"; document.getElementById('pCredit').innerText = (data.creditScoreInfo && data.creditScoreInfo.creditScore) ? data.creditScoreInfo.creditScore : "N/A"; document.getElementById('pClan').innerText = (data.clanBasicInfo && data.clanBasicInfo.clanName) ? data.clanBasicInfo.clanName : "No Clan"; document.getElementById('pSignature').innerText = (data.socialInfo && data.socialInfo.signature) ? data.socialInfo.signature : "No Bio"; loader.style.display = 'none'; resultCard.style.display = 'block'; } else { throw new Error("Invalid UID or Profile Not Found"); } } catch (e) { loader.style.display = 'none'; errorMsg.innerText = "Error: " + e.message; errorMsg.style.display = 'block'; } });
